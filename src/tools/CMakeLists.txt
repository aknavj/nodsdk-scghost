cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(nodsdk-tools C CXX)
    if(WIN32)
        enable_language(RC)
    endif()
endif()
if(NOT TARGET nodsdk_app_icon)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../res ${CMAKE_CURRENT_BINARY_DIR}/resources)
endif()

option(NODSDK_BUILD_EDITORS "Build native NSD and NOT tree editors (ImGui/GLFW)" OFF)
if(NOT TARGET asset_parsers)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../parsers ${CMAKE_CURRENT_BINARY_DIR}/asset_parsers)
endif()
add_library(nod_tool_cli STATIC common/tool_cli.cpp)
target_include_directories(nod_tool_cli PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/common
    PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/editors)
target_compile_features(nod_tool_cli PUBLIC cxx_std_17)
if(WIN32)
    target_link_libraries(nod_tool_cli PUBLIC shell32)
endif()
add_subdirectory(editors)

add_executable(NILtoMAP nil2map/niltomap.c)
add_library(nil_map STATIC common/niltomap.c)
target_include_directories(nil_map PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/common)
target_link_libraries(NILtoMAP PRIVATE nil_map)
add_library(nod_scene_bridge STATIC
    common/scene_bridge.cpp
    common/path_graph.cpp
    common/map_lighting.cpp
    common/tool_common.c
    common/nod_assets.c
    common/nod_native.cpp
    common/nod_materials.c
    nil2map/nodscene_export.cpp
    map2nil/nodscene_compile.cpp)
target_link_libraries(nod_scene_bridge PUBLIC nil_map asset_parsers nod_tool_cli)
target_include_directories(nod_scene_bridge PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/common)
target_compile_features(nod_scene_bridge PUBLIC cxx_std_17)
add_executable(NODscene common/nodscene_cli.cpp)
target_link_libraries(NODscene PRIVATE nod_scene_bridge)
add_library(nmb_compiler STATIC common/nmb_compiler.cpp)
add_library(mopp_compiler STATIC common/mopp_compiler.cpp)
target_include_directories(mopp_compiler PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/common)
target_compile_features(mopp_compiler PUBLIC cxx_std_17)
add_library(world_geometry STATIC common/world_geometry.cpp common/world_render.cpp common/world_sector.cpp common/world_image.cpp common/world_lighting.cpp)
target_link_libraries(world_geometry PUBLIC nod_scene_bridge mopp_compiler)
target_compile_features(world_geometry PUBLIC cxx_std_17)
target_include_directories(nmb_compiler PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/common)
target_link_libraries(nmb_compiler PUBLIC asset_parsers)
add_executable(NODtoNMB nod2nmb/nodtonmb.cpp)
target_link_libraries(NODtoNMB PRIVATE nmb_compiler nod_editor_document)
add_executable(MAPtoNIL map2nil/maptonil.cpp)
target_link_libraries(MAPtoNIL PRIVATE world_geometry nod_editor_document)
add_executable(MAPtoNSD map2nil/maptonsd.cpp)
target_link_libraries(MAPtoNSD PRIVATE nod_scene_bridge nod_editor_document)
add_executable(MAPtoNPD map2nil/maptonpd.cpp)
target_link_libraries(MAPtoNPD PRIVATE nod_scene_bridge nod_editor_document)
foreach(target nmb_compiler NODtoNMB MAPtoNIL MAPtoNSD MAPtoNPD)
    target_compile_features(${target} PUBLIC cxx_std_17)
endforeach()
set_target_properties(NODscene nod_scene_bridge PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
foreach(tool NILtoMAP NODscene NODtoNMB MAPtoNIL MAPtoNSD MAPtoNPD)
    target_link_libraries(${tool} PRIVATE nodsdk_app_icon)
endforeach()
foreach(tool NILtoMAP nil_map NODscene nod_scene_bridge nod_tool_cli nmb_compiler NODtoNMB MAPtoNIL MAPtoNSD MAPtoNPD mopp_compiler world_geometry)
    set_target_properties(${tool} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${tool} PRIVATE /W4)
        target_compile_definitions(${tool} PRIVATE _CRT_SECURE_NO_WARNINGS)
        if(DEFINED ENV{CI})
            target_compile_options(${tool} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${tool} PRIVATE -Wall -Wextra -Wpedantic)
        target_link_libraries(${tool} PRIVATE m)
        if(DEFINED ENV{CI})
            target_compile_options(${tool} PRIVATE -Werror)
        endif()
    endif()
endforeach()
